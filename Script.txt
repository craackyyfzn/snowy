local dragging, dragInput, dragStart, startPos

local function startDragging(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end

MainFrame.InputBegan:Connect(startDragging)
TopBar.InputBegan:Connect(startDragging)

MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

local gravityData = {}
local footParticles

local function removeSnowyGravity(root, hum)
	if footParticles then footParticles:Destroy() footParticles = nil end
	if root then
		local f = root:FindFirstChild("GravityCoilEffect")
		if f then f:Destroy() end
	end
	if hum and gravityData.oldJumpPower ~= nil then
		if gravityData.useJumpPower then
			hum.JumpPower = gravityData.oldJumpPower
		else
			hum.JumpHeight = gravityData.oldJumpHeight
		end
	end
	if gravityData.heartbeat then gravityData.heartbeat:Disconnect() end
	if gravityData.runConnection then gravityData.runConnection:Disconnect() end
	if gravityData.diedConnection then gravityData.diedConnection:Disconnect() end
	gravityData = {}
end

local function applySnowyGravity(active, root, hum, char, tool)
	if active then
		removeSnowyGravity(root, hum)
		if not root or not hum then return end

		local force = Instance.new("BodyForce")
		force.Name = "GravityCoilEffect"
		force.Parent = root
		force.Force = Vector3.new(0, root.AssemblyMass * workspace.Gravity * Config.GravityReduction, 0)

		gravityData.oldJumpPower = hum.JumpPower
		gravityData.oldJumpHeight = hum.JumpHeight
		gravityData.useJumpPower = hum.UseJumpPower
		if hum.UseJumpPower then
			hum.JumpPower = Config.JumpPowerR6
		else
			hum.JumpHeight = Config.JumpHeightR15
		end

		footParticles = Instance.new("ParticleEmitter")
		footParticles.Name = "SnowFootsteps"
		footParticles.Texture = "rbxassetid://152984220"
		footParticles.Rate = 0
		footParticles.Lifetime = NumberRange.new(0.5, 1)
		footParticles.Speed = NumberRange.new(5, 10)
		footParticles.SpreadAngle = Vector2.new(180, 180)
		footParticles.Size = NumberSequence.new({
			NumberSequenceKeypoint.new(0, 0.4),
			NumberSequenceKeypoint.new(1, 0)
		})
		footParticles.Transparency = NumberSequence.new({
			NumberSequenceKeypoint.new(0, 0.5),
			NumberSequenceKeypoint.new(1, 1)
		})
		footParticles.Color = ColorSequence.new(Color3.fromRGB(255, 255, 255))
		footParticles.Parent = root

		gravityData.heartbeat = RunService.Heartbeat:Connect(function()
			if not tool.Parent or tool.Parent ~= char or not char.Parent or not hum.Parent or not root.Parent then
				removeSnowyGravity(root, hum)
				return
			end
			force.Force = Vector3.new(0, root.AssemblyMass * workspace.Gravity * Config.GravityReduction, 0)
		end)

		gravityData.runConnection = hum.Running:Connect(function(speed)
			if tool.Parent == char and speed > 1 then
				if footParticles then footParticles.Rate = Config.ParticlesRateRunning end
			else
				if footParticles then footParticles.Rate = 0 end
			end
		end)

		gravityData.diedConnection = hum.Died:Connect(function()
			removeSnowyGravity(root, hum)
		end)
	else
		removeSnowyGravity(root, hum)
	end
end

local function buildSnowyTool()
	local Tool = Instance.new("Tool")
	Tool.Name = "SnowyCoil"
	Tool.CanBeDropped = false
	Tool.RequiresHandle = true
	Tool.TextureId = Config.IconId
	Tool.Grip = CFrame.new(0, 0, 0) * CFrame.Angles(math.rad(90), 0, 0)

	local Handle = Instance.new("Part", Tool)
	Handle.Name = "Handle"
	Handle.Size = Vector3.new(1, 1.2, 2)
	Handle.CanCollide, Handle.Massless, Handle.CanTouch, Handle.CanQuery = false, true, false, false

	local Mesh = Instance.new("SpecialMesh", Handle)
	Mesh.MeshType = Enum.MeshType.FileMesh
	Mesh.MeshId = Config.MeshId
	Mesh.TextureId = Config.MeshTexture
	Mesh.Scale = Vector3.new(1.4, 1.4, 1.4)

	local Sound = Instance.new("Sound", Handle)
	Sound.SoundId = Config.SoundId
	Sound.Volume = 1

	Tool.Equipped:Connect(function()
		Sound:Play()
		local char = LocalPlayer.Character
		if char then
			applySnowyGravity(true, char:FindFirstChild("HumanoidRootPart"), char:FindFirstChildOfClass("Humanoid"), char, Tool)
		end
	end)

	Tool.Unequipped:Connect(function()
		Sound:Stop()
		local char = LocalPlayer.Character
		if char then
			applySnowyGravity(false, char:FindFirstChild("HumanoidRootPart"), char:FindFirstChildOfClass("Humanoid"), char, Tool)
		end
	end)

	return Tool
end

local function giveSnowyTool()
    if not LocalPlayer.Backpack:FindFirstChild("SnowyCoil") and not (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("SnowyCoil")) then
        buildSnowyTool().Parent = LocalPlayer.Backpack
    end
end

local snowyToggled = false

local function toggleSnowyCoil()
    snowyToggled = not snowyToggled
    if snowyToggled then
        TweenService:Create(ToggleButton, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(0, 180, 255)}):Play()
        TweenService:Create(StatusDot, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(0, 255, 120)}):Play()
        ToggleStroke.Color = Color3.fromRGB(100, 220, 255)
        ToggleButton.TextColor3 = Color3.fromRGB(255, 255, 255)
        ToggleButton.Text = "ON"
        pcall(function() giveSnowyTool() end)
    else
        TweenService:Create(ToggleButton, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(24, 24, 34)}):Play()
        TweenService:Create(StatusDot, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(80, 80, 95)}):Play()
        ToggleStroke.Color = Color3.fromRGB(45, 45, 60)
        ToggleButton.TextColor3 = Color3.fromRGB(120, 120, 135)
        ToggleButton.Text = "OFF"
        
        pcall(function()
            local char = LocalPlayer.Character
            if char then
                removeSnowyGravity(char:FindFirstChild("HumanoidRootPart"), char:FindFirstChildOfClass("Humanoid"))
            end
            for _, loc in ipairs({LocalPlayer.Character, LocalPlayer.Backpack}) do
                if loc then
                    for _, item in ipairs(loc:GetChildren()) do
                        local name = item.Name:lower()
                        if name:find("snowy") then item:Destroy() end
                    end
                end
            end
        end)
    end
end

local isMobile = UserInputService.TouchEnabled and not UserInputService.MouseEnabled

if isMobile then
    ToggleButton.TouchTap:Connect(toggleSnowyCoil)
else
    ToggleButton.MouseButton1Click:Connect(toggleSnowyCoil)
end

LocalPlayer.CharacterAdded:Connect(function()
    task.wait(0.5)
    if snowyToggled then giveSnowyTool() end
end)
